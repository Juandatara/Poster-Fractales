\section{Pseudocódigo}

Para generar el tetraedro de Sierspinski se usó el siguiente algoritmo, basado en el triángulo de Sierspinski en \cite{Diego}.

\renewcommand\algorithmicrequire{\textbf{Input:}}
\renewcommand\algorithmicensure{\textbf{Output:}}

\begin{algorithm}[H]
	\caption{Tetraedro de Sierspinski de longitud $L$ con $n$ iteraciones.}
	\begin{algorithmic}
		\REQUIRE Longitud arista ($L$), Punto inicial ($P_0$), iteraciones ($n$).
		\ENSURE Tetraedro de Sierspinki
		\IF{$n == 0$}
			\STATE $P_1 \gets [P_0[0]+L, \ \ P_0[1] \ \ ,P_0[2]]$
			\STATE $P_2 \gets [P_0[0]+L/2, \ \ P_0[1]+\sqrt{3}L/2, \ \ P_0[2]]$
			\STATE $P_3 \gets [P_0[0]+L/2, \ \ P_0[1]+\sqrt{3}L/6, \ \ P_0[2]+\sqrt{6}L/3]$
			\STATE TetraedroRegular($P_0,P_1,P_2,P_3$)
		\ELSE
			\STATE TetraedroSierpinski($L/2, \ \ P_0, \ \ n-1$)
			\STATE TetraedroSierpinski($L/2, \ \ [P_0[0]+L/2, \ \ P_0[1], \ \ P_0[2]], \ \ n-1$)
			\STATE TetraedroSierpinski($L/2, \ \ [P_0[0]+L/4, \ \ P_0[1]+\sqrt{3}L/4, \ \ P_0[2]], \ \ n-1$)
			\STATE TetraedroSierpinski($L/2, \ \ [P_0[0]+L/4, \ \ P_0[1]+\sqrt{3}L/12, \ \ P_0[2]+\sqrt{6}L/6], \ \ n-1$)
		\ENDIF
	\end{algorithmic}
\end{algorithm}
